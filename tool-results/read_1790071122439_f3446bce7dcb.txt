    47→

    56→
    57→  // OCR tab state

    60→  const [isDraggingImage, setIsDraggingImage] = useState(false);

    62→  const [ocrProgress, setOcrProgress] = useState<OcrProgress | null>(null);
    63→  const [ocrText, setOcrText] = useState<string>("");
    64→  const [ocrConfidence, setOcrConfidence] = useState<number | null>(null);
    65→
    66→  // Revoke object URLs on unmount / when replaced

    76→      const url = `https://github.com/${githubRepo}/archive/refs/heads/main.zip`;

    92→          variant: "destructive",

    97→      const url = URL.createObjectURL(blob);
    98→      const a = document.createElement("a");
    99→      a.href = url;
   100→      a.download = "lengger-ledger-converter.zip";
   101→      document.body.appendChild(a);
   102→      a.click();
   103→      document.body.removeChild(a);
   104→      URL.revokeObjectURL(url);
   105→      toast({
   106→        title: "Source code terunduh",
   107→        description: "lengger-ledger-converter.zip — versi terbaru.",

   109→    } catch (err) {
   110→      const msg = err instanceof Error ? err.message : "Unknown error";
   111→      toast({
   112→        title: "Download source gagal",
   113→        description: msg,
   114→        variant: "destructive",

   117→      setIsDownloadingSource(false);
   118→    }
   119→  }, []);
   120→
   121→  // ---------- File upload handler (existing flow) ----------
   122→  const handleFile = useCallback(async (file: File) => {
   123→    const lower = file.name.toLowerCase();
   124→    if (!lower.endsWith(".md") && !lower.endsWith(".csv")) {
   125→      toast({
   126→        title: "Format tidak didukung",
   127→        description: "File harus berekstensi .md atau .csv",
   128→        variant: "destructive",
   129→      });
   130→      return;
   131→    }
   132→    const raw = await file.text();
   133→    const { normalizeInput } = await import("@/lib/budaya/normalize-input");
   134→    const { mdText, format } = normalizeInput(raw, file.name);
   135→    setFileName(file.name);
   136→    setFileContent(mdText);
   137→    const result = parseMd(mdText, file.name);
   138→    setPreview({
   139→      filename: file.name,
   140→      tanggalList: result.tanggalList,
   141→      rows: result.rows,
   142→      warnings: result.warnings,
   143→    });
   144→    if (format === "fb") {
   145→      toast({
   146→        title: "Format FB scraping terdeteksi",
   147→        description: `${result.rows.length} entri diparsing dari post FB.`,
   148→      });
   149→    } else if (format === "csv") {
   150→      toast({
   151→        title: "Format CSV scraping terdeteksi",
   152→        description: `${result.rows.length} entri diparsing dari CSV.`,
   153→      });
   154→    }
   155→  }, []);
   156→
   157→  const onInputChange = useCallback(
   158→    async (e: React.ChangeEvent<HTMLInputElement>) => {

   160→      if (f) await handleFile(f);
   161→    },
   162→    [handleFile]
   163→  );
   164→
   165→  const onDrop = useCallback(
   166→    async (e: React.DragEvent<HTMLDivElement>) => {
   167→      e.preventDefault();
   168→      setIsDragging(false);
   169→      const f = e.dataTransfer.files?.[0];
   170→      if (f) await handleFile(f);
   171→    },
   172→    [handleFile]
   173→  );
   174→
   175→  // ---------- OCR tab handlers ----------
   176→  const addImages = useCallback(
   177→    (files: FileList | File[]) => {
   178→      const arr = Array.from(files).filter((f) => {
   179→        const t = f.type.toLowerCase();
   180→        return (
   181→          t.startsWith("image/") ||
   182→          /\.(png|jpe?g|gif|webp|bmp)$/i.test(f.name)
   183→        );
   184→      });
   185→      if (arr.length === 0) {
   186→        toast({
   187→          title: "Tidak ada gambar valid",
   188→          description: "Format didukung: PNG, JPEG, GIF, WebP, BMP.",
   189→          variant: "destructive",
   190→        });
   191→        return;
   192→      }
   193→      // Revoke old previews
   194→      ocrPreviews.forEach((url) => URL.revokeObjectURL(url));
   195→      const newPreviews = arr.map((f) => URL.createObjectURL(f));
   196→      setOcrImages((prev) => [...prev, ...arr]);
   197→      setOcrPreviews((prev) => [...prev, ...newPreviews]);
   198→      // Reset previous OCR result when adding new images
   199→      setOcrText("");
   200→      setOcrConfidence(null);
   201→    },
   202→    [ocrPreviews]
   203→  );
   204→
   205→  const onImageInputChange = useCallback(
   206→    (e: React.ChangeEvent<HTMLInputElement>) => {
   207→      if (e.target.files && e.target.files.length > 0) {
   208→        addImages(e.target.files);
   209→      }
   210→      // reset so selecting the same file again still triggers change
   211→      if (imageInputRef.current) imageInputRef.current.value = "";
   212→    },
   213→    [addImages]
   214→  );
   215→
   216→  const onImageDrop = useCallback(
   217→    (e: React.DragEvent<HTMLDivElement>) => {
   218→      e.preventDefault();
   219→      setIsDraggingImage(false);
   220→      if (e.dataTransfer.files && e.dataTransfer.files.length > 0) {
   221→        addImages(e.dataTransfer.files);
   222→      }
   223→    },
   224→    [addImages]
   225→  );
   226→
   227→  const removeImage = useCallback(
   228→    (index: number) => {
   229→      setOcrImages((prev) => prev.filter((_, i) => i !== index));
   230→      setOcrPreviews((prev) => {
   231→        const url = prev[index];
   232→        if (url) URL.revokeObjectURL(url);
   233→        return prev.filter((_, i) => i !== index);
   234→      });
   235→    },
   236→    []
   237→  );
   238→
   239→  const clearImages = useCallback(() => {
   240→    ocrPreviews.forEach((url) => URL.revokeObjectURL(url));
   241→    setOcrImages([]);
   242→    setOcrPreviews([]);
   243→    setOcrText("");
   244→    setOcrConfidence(null);
   245→    setOcrProgress(null);
   246→  }, [ocrPreviews]);
   247→
   248→  const runOcrOnImages = useCallback(async () => {
   249→    if (ocrImages.length === 0) {
   250→      toast({
   251→        title: "Belum ada gambar",
   252→        description: "Tambahkan minimal 1 gambar untuk OCR.",
   253→        variant: "destructive",
   254→      });
   255→      return;
   256→    }
   257→    setIsOcrRunning(true);
   258→    setOcrProgress({ status: "Memulai...", progress: 0 });
   259→    setOcrText("");
   260→    setOcrConfidence(null);
   261→
   262→    try {
   263→      const result = await runOcrBatch(ocrImages, "ind+eng", (p) => {
   264→        setOcrProgress(p);
   265→      });
   266→      setOcrText(result.text);
   267→      const avgConf =
   268→        result.items.length > 0
   269→          ? result.items.reduce((sum, it) => sum + it.confidence, 0) /
   270→            result.items.length
   271→          : 0;
   272→      setOcrConfidence(avgConf);
   273→
   274→      const looksValid = looksLikeInfoLengger(result.text);
   275→      toast({
   276→        title: result.text.length > 0 ? "OCR selesai" : "OCR hasil kosong",
   277→        description: `${result.items.length} gambar diproses · confidence ${avgConf.toFixed(
   278→          1
   279→        )}%${
   280→          !looksValid && result.text.length > 0
   281→            ? " · ⚠ teks tidak terdeteksi sebagai Info Lengger"
   282→            : ""
   283→        }`,
   284→        variant: result.text.length > 0 ? "default" : "destructive",
   285→      });
   286→    } catch (err) {
   287→      const msg = err instanceof Error ? err.message : "Unknown OCR error";
   288→      toast({
   289→        title: "OCR gagal",
   290→        description: msg,
   291→        variant: "destructive",
   292→      });
   293→    } finally {
   294→      setIsOcrRunning(false);
   295→      setOcrProgress(null);
   296→    }
   297→  }, [ocrImages]);
   298→
   299→  // Feed OCR text into the existing parser pipeline
   300→  const useOcrAsMd = useCallback(() => {
   301→    if (!ocrText.trim()) {
   302→      toast({
   303→        title: "Teks OCR kosong",
   304→        description: "Jalankan OCR dulu atau ketik manual.",
   305→        variant: "destructive",
   306→      });
   307→      return;
   308→    }
   309→    // Build a sensible filename from the first image name
   310→    const baseName =
   311→      ocrImages.length > 0
   312→        ? ocrImages[0].name.replace(/\.[^.]+$/, "")
   313→        : "ocr-result";
   314→    const derivedName = `${baseName}.md`;
   315→    setFileName(derivedName);
   316→    setFileContent(ocrText);
   317→    const result = parseMd(ocrText, derivedName);
   318→    setPreview({
   319→      filename: derivedName,
   320→      tanggalList: result.tanggalList,
   321→      rows: result.rows,
   322→      warnings: result.warnings,
   323→    });
   324→    toast({
   325→      title: "Teks OCR dipakai sebagai input",
   326→      description: `${result.rows.length} entri terdeteksi. Scroll ke bawah untuk preview.`,
   327→    });
   328→  }, [ocrText, ocrImages]);
   329→
   330→  // Download OCR text as .md file — user can edit offline, then re-upload
   331→  // via the "File .md / .csv" tab and convert to .xlsx using the existing pipeline.
   332→  const downloadOcrAsMd = useCallback(() => {
   333→    if (!ocrText.trim()) {
   334→      toast({
   335→        title: "Teks OCR kosong",
   336→        description: "Jalankan OCR dulu atau ketik manual sebelum download.",
   337→        variant: "destructive",
   338→      });
   339→      return;
   340→    }
   341→    // Build a sensible filename from the first image name (or fallback to "ocr-result")
   342→    const baseName =
   343→      ocrImages.length > 0
   344→        ? ocrImages[0].name.replace(/\.[^.]+$/, "")
   345→        : "ocr-result";
   346→    const outName = `${baseName}.md`;
   347→    const blob = new Blob([ocrText], { type: "text/markdown;charset=utf-8" });
   348→    const url = URL.createObjectURL(blob);
   349→    const a = document.createElement("a");
   350→    a.href = url;
   351→    a.download = outName;
   352→    document.body.appendChild(a);
   353→    a.click();
   354→    document.body.removeChild(a);
   355→    URL.revokeObjectURL(url);
   356→    toast({
   357→      title: "File .md terunduh",
   358→      description: `${outName} (${ocrText.length.toLocaleString()} karakter). Edit di editor favoritmu, lalu upload via tab "File .md / .csv" untuk convert ke .xlsx.`,
   359→    });
   360→  }, [ocrText, ocrImages]);
   361→
   362→  // ---------- Convert & reset handlers ----------
   363→  const onConvert = useCallback(async () => {
   364→    if (!preview || !fileName) return;
   365→    setIsConverting(true);
   366→    try {
   367→      const { generateXlsx } = await import("@/lib/budaya/xlsx-generator");
   368→
   369→      // If fileContent exists (from RAW): re-parse for fresh xlsx
   370→      // If null (from Excel): use preview.rows directly (already parsed from Excel)
   371→      let rows = preview.rows;
   372→      let warnings: string[] = [];
   373→      if (fileContent) {
   374→        const { normalizeInput } = await import("@/lib/budaya/normalize-input");
   375→        const { mdText } = normalizeInput(fileContent, fileName);
   376→        const parsed = parseMd(mdText, fileName);
   377→        rows = parsed.rows;
   378→        warnings = parsed.warnings;
   379→      }
   380→
   381→      if (rows.length === 0) {
   382→        toast({
   383→          title: "Konversi gagal",
   384→          description: "Tidak ada entri terdeteksi. Cek format input (mis. harus ada header 'Info Lengger' atau '<n>_<lokasi>').",
   385→          variant: "destructive",
   386→        });
   387→        return;
   388→      }
   389→
   390→      const arrayBuffer = await generateXlsx(rows);
   391→      const blob = new Blob([arrayBuffer], {
   392→        type: "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
   393→      });
   394→
   395→      const baseName = fileName.replace(/\.(md|csv)$/i, "");
   396→      const outName = `${baseName}.xlsx`;
   397→      const url = URL.createObjectURL(blob);
   398→      const a = document.createElement("a");
   399→      a.href = url;
   400→      a.download = outName;
   401→      document.body.appendChild(a);
   402→      a.click();
   403→      document.body.removeChild(a);
   404→      URL.revokeObjectURL(url);
   405→
   406→      const warnCount = warnings.length;
   407→      toast({
   408→        title: "Konversi berhasil",
   409→        description: `${rows.length} baris diproses${
   410→          warnCount > 0 ? ` · ${warnCount} catatan` : ""
   411→        }. File ${outName} diunduh.`,
   412→      });
   413→    } catch (err) {
   414→      const msg = err instanceof Error ? err.message : "Unknown error";
   415→      toast({
   416→        title: "Konversi gagal",
   417→        description: msg,
   418→        variant: "destructive",
   419→      });
   420→    } finally {
   421→      setIsConverting(false);
   422→    }
   423→  }, [fileContent, fileName]);
   424→
   425→  // Download raw-clean.md (single file, all days concatenated)
   426→  const onDownloadRawCleanMd = useCallback(async () => {
   427→    if (!fileContent || !fileName) return;
   428→    setIsGeneratingMd(true);
   429→    try {
   430→      const { generateCleanMdSingleFile } = await import("@/lib/budaya/clean-md-generator");
   431→      const mdText = generateCleanMdSingleFile(fileContent, fileName);
   432→      const blob = new Blob([mdText], { type: "text/markdown;charset=utf-8" });
   433→
   434→      const baseName = fileName.replace(/\.(md|csv)$/i, "");
   435→      const outName = `${baseName}-raw-clean.md`;
   436→      const url = URL.createObjectURL(blob);
   437→      const a = document.createElement("a");
   438→      a.href = url;
   439→      a.download = outName;
   440→      document.body.appendChild(a);
   441→      a.click();
   442→      document.body.removeChild(a);
   443→      URL.revokeObjectURL(url);
   444→
   445→      toast({
   446→        title: "Raw clean MD terunduh",
   447→        description: `${outName} — 1 file .md (semua tanggal, format Info Lengger standar).`,
   448→      });
   449→    } catch (err) {
   450→      const msg = err instanceof Error ? err.message : "Unknown error";
   451→      toast({
   452→        title: "Generate raw clean MD gagal",
   453→        description: msg,
   454→        variant: "destructive",
   455→      });
   456→    } finally {
   457→      setIsGeneratingMd(false);
   458→    }
   459→  }, [fileContent, fileName]);
   460→
   461→  // Upload edited Excel → parse → preview
   462→  const handleExcelUpload = useCallback(async (file: File) => {
   463→    const lower = file.name.toLowerCase();
   464→    if (!lower.endsWith(".xlsx")) {
   465→      toast({
   466→        title: "Format tidak didukung",
   467→        description: "File harus berekstensi .xlsx",
   468→        variant: "destructive",
   469→      });
   470→      return;
   471→    }
   472→    setIsExcelParsing(true);
   473→    try {
   474→      const { parseXlsx } = await import("@/lib/budaya/xlsx-parser");
   475→      const { rows, warnings } = await parseXlsx(file);
   476→
   477→      if (rows.length === 0) {
   478→        toast({
   479→          title: "Excel kosong",
   480→          description: "Tidak ada baris data terdeteksi.",
   481→          variant: "destructive",
   482→        });
   483→        return;
   484→      }
   485→
   486→      setFileName(file.name);
   487→      setFileContent(null); // Excel is binary, not text
   488→      const tanggalList = [...new Set(rows.map((r) => r.Tanggal).filter(Boolean))] as string[];
   489→      setPreview({
   490→        filename: file.name,
   491→        tanggalList,
   492→        rows: rows as RowData[],
   493→        warnings,
   494→      });
   495→
   496→      toast({
   497→        title: "Excel terbaca",
   498→        description: `${rows.length} baris diparsing dari Excel.`,
   499→      });
   500→    } catch (err) {
   501→      const msg = err instanceof Error ? err.message : "Unknown error";
   502→      toast({
   503→        title: "Baca Excel gagal",
   504→        description: msg,
   505→        variant: "destructive",
   506→      });
   507→    } finally {
   508→      setIsExcelParsing(false);
   509→    }
   510→  }, []);
   511→
   512→  const onExcelInputChange = useCallback(
   513→    async (e: React.ChangeEvent<HTMLInputElement>) => {
   514→      const f = e.target.files?.[0];
   515→      if (f) await handleExcelUpload(f);
   516→      if (excelInputRef.current) excelInputRef.current.value = "";
   517→    },
   518→    [handleExcelUpload]
   519→  );
   520→
   521→  // Download weekly .xlsx ZIP (1 file per ISO week, Senin-Minggu)

   525→    try {
   526→      const { generateWeeklyXlsxZip } = await import("@/lib/budaya/weekly-xlsx-generator");
   527→      const baseName = fileName.replace(/\.(md|csv|xlsx)$/i, "");

   529→      const blob = new Blob([arrayBuffer], { type: "application/zip" });
   530→      const outName = `${baseName}-weekly.zip`;
   531→      const url = URL.createObjectURL(blob);
   532→      const a = document.createElement("a");
   533→      a.href = url;
   534→      a.download = outName;
   535→      document.body.appendChild(a);
   536→      a.click();
   537→      document.body.removeChild(a);
   538→      URL.revokeObjectURL(url);
   539→      toast({
   540→        title: "Weekly Excel ZIP terunduh",
   541→        description: `${outName} — 1 file .xlsx per minggu (ISO week). Edit tiap minggu, lalu upload via tab Excel.`,
   542→      });
   543→    } catch (err) {
   544→      const msg = err instanceof Error ? err.message : "Unknown error";
   545→      toast({
   546→        title: "Generate weekly Excel gagal",
   547→        description: msg,
   548→        variant: "destructive",
   549→      });
   550→    } finally {
   551→      setIsGeneratingWeekly(false);
   552→    }
   553→  }, [preview, fileName]);
   554→
   555→  // Download clean.md ZIP from edited Excel
   556→  const onDownloadExcelCleanMd = useCallback(async () => {
   557→    if (!preview || !fileName) return;
   558→    setIsGeneratingExcelMd(true);
   559→    try {
   560→      const { generateCleanMdZipFromRows } = await import("@/lib/budaya/clean-md-generator");
   561→      const arrayBuffer = await generateCleanMdZipFromRows(preview.rows, fileName);
   562→      const blob = new Blob([arrayBuffer], { type: "application/zip" });
